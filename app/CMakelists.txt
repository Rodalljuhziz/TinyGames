function(copy_mingw_dlls target_name)
    foreach(config IN ITEMS Debug Release)
        foreach(dll IN LISTS MINGW_DLLS)
            add_custom_command(TARGET ${target_name} POST_BUILD
                    COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "${MINGW_DLL_DIR}/${dll}"
                    "$<TARGET_FILE_DIR:${target_name}>/${dll}"
                    COMMENT "Copying ${dll} to ${config} output directory"
            )
        endforeach()
    endforeach()
endfunction()

copy_mingw_dlls(${TARGET_NAME})
